Aim :

Implement asynchronous background processing using Node.js native EventEmitter. When a task is created or deleted, emit an event handled by a notification listener without blocking the main API response.

Code :

backend/events.js :-

const EventEmitter = require('events');

class TaskEvents extends EventEmitter {}

module.exports = new TaskEvents();

backend/listeners.js :-

const taskEvents = require('./events');

function delay(milliseconds) {
  return new Promise((resolve) => setTimeout(resolve, milliseconds));
}

taskEvents.on('task-created', (task) => {
  setImmediate(async () => {
    try {
      await delay(2000);
      console.log(`[Notification] Task "${task.title}" created at ${new Date().toISOString()}`);
    } catch (error) {
      taskEvents.emit('error', error);
    }
  });
});

taskEvents.on('task-deleted', (task) => {
  setImmediate(() => {
    console.log(`[Notification] Task "${task.title}" deleted at ${new Date().toISOString()}`);
  });
});

taskEvents.on('error', (error) => {
  console.error(`[Notification error] ${error.message}`);
});

backend/routes/tasks.js event usage :-

const taskEvents = require('../events');

router.post('/', asyncHandler(async (req, res) => {
  const task = await Task.create(req.body);
  cache.invalidateTaskCaches();
  res.status(201).json(task);
  taskEvents.emit('task-created', task);
}));

router.delete('/:id', ensureValidId, asyncHandler(async (req, res) => {
  const task = await Task.findByIdAndDelete(req.params.id);
  if (!task) return res.status(404).json({ error: 'Task not found' });
  cache.invalidateTaskCaches(req.params.id);
  res.status(200).json(task);
  taskEvents.emit('task-deleted', task);
}));
