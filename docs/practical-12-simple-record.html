Aim :

Set up a GitHub Actions CI/CD pipeline for the Task Management application. The pipeline must install dependencies, run automated backend tests, build the frontend, and report a clear pass or fail result on pushes and pull requests targeting the main branch.

Code :

.github/workflows/ci.yml :-

name: CI Pipeline

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  backend-test:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
          cache-dependency-path: backend/package-lock.json
      - run: npm ci
      - run: npm test

  frontend-build:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: frontend
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
          cache-dependency-path: frontend/package-lock.json
      - run: npm ci
      - run: npm run build

backend/test/cache.test.js :-

const cache = require('../cache');

describe('task cache', () => {
  beforeEach(() => {
    cache.invalidateTaskCaches();
  });

  test('stores and returns a cached value', () => {
    const tasks = [{ _id: 'task-1', title: 'CI task' }];
    cache.set('tasks:all', tasks);
    expect(cache.get('tasks:all')).toEqual(tasks);
  });

  test('invalidates collection and single-task keys', () => {
    cache.set('tasks:all', [{ _id: 'task-1' }]);
    cache.set('tasks:task-1', { _id: 'task-1' });
    cache.invalidateTaskCaches('task-1');
    expect(cache.get('tasks:all')).toBeUndefined();
    expect(cache.get('tasks:task-1')).toBeUndefined();
  });
});
